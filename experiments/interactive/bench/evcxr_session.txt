:timing
:dep serde_json = "1"
:dep regex = "1"
let json = r#"[{"id":1,"customer":"ann@example.com","total_cents":1250},{"id":2,"customer":"bob@example.com","total_cents":999}]"#;
let orders: Vec<serde_json::Value> = serde_json::from_str(json).expect("fixture JSON");
let re: regex::Regex = regex::Regex::new(r"^[^@\s]+@[^@\s]+\.[a-z]+$").expect("static regex");
println!("{} orders", orders.len());
for o in &orders { println!("{} valid={}", o["customer"], re.is_match(o["customer"].as_str().expect("customer"))); }
let total: i64 = orders.iter().map(|o| o["total_cents"].as_i64().expect("total_cents")).sum(); println!("{total}");
println!("{}", serde_json::to_string_pretty(&orders[0]).expect("serialize JSON"));
println!("{}", orders.len() + 1);
println!("{}", orders.len() + 2);
